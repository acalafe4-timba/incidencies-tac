    <script type="text/babel">

        const { useState, useEffect, useRef } = React;


        /**

         * Component d'Icona que utilitza Lucide.

         * Requereix cridar lucide.createIcons() després de cada renderitzat.

         */

        const Icon = ({ name, size = 24, className = "" }) => (

            <i

                data-lucide={name.toLowerCase().replace(/([a-z0-9])([A-Z])/g, '$1-$2')}

                style={{ width: size, height: size, display: 'inline-block' }}

                className={className}

            ></i>

        );


        function App() {

            // Estats de l'aplicació

            const [user, setUser] = useState(null);

            const [incidencies, setIncidencies] = useState([]);

            const [view, setView] = useState('list');

            const [loading, setLoading] = useState(true);

           

            // Estats del formulari

            const [docent, setDocent] = useState('');

            const [descripcio, setDescripcio] = useState('');

            const [aula, setAula] = useState('');

            const [prioritat, setPrioritat] = useState('Mitjana');

            const [tipus, setTipus] = useState('Hardware');

            const [imageData, setImageData] = useState(null);

            const [isSubmitting, setIsSubmitting] = useState(false);

           

            // Estats de la Intel·ligència Artificial (Gemini)

            const [isAiLoading, setIsAiLoading] = useState(false);

            const [aiSuggestion, setAiSuggestion] = useState({ id: null, text: '' });


            const fileInputRef = useRef(null);


            // Refrescar icones de Lucide quan el DOM canvia

            useEffect(() => {

                if (window.lucide) window.lucide.createIcons();

            });


            // Inicialització de l'autenticació

            useEffect(() => {

                const initAuth = async () => {

                    if (!window.fb) return;

                    try {

                       await window.fb.signInAnonymously(window.fb.auth);

                        }

                    } catch (err) { console.error("Auth error:", err); }

                };

                initAuth();

                const unsubscribe = window.fb.onAuthStateChanged(window.fb.auth, (u) => {

                    setUser(u);

                    if (u) setLoading(false);

                });

                return () => unsubscribe();

            }, []);


            // Subscripció a les dades de Firestore

            useEffect(() => {

                if (!user || !window.fb) return;

                const q = window.fb.collection(window.fb.db, 'artifacts', window.fb.appId, 'public', 'data', 'incidencies');

                const unsubscribe = window.fb.onSnapshot(q, (snapshot) => {

                    const data = snapshot.docs.map(doc => ({

                        id: doc.id,

                        ...doc.data()

                    })).sort((a, b) => (b.createdAt?.seconds || 0) - (a.createdAt?.seconds || 0));

                    setIncidencies(data);

                }, (error) => console.error("Error al carregar dades:", error));

                return () => unsubscribe();

            }, [user]);


            /**

             * Funció genèrica per cridar a l'API de Gemini amb backoff exponencial

             */

            const callGemini = async (prompt, systemInstruction) => {

                const apiKey = "AIzaSyAoNfxx3rq0OUaF6V1BfAPCGtzYhLkbq4I"; // La clau s'injecta automàticament a l'entorn de l'app

                let retries = 0;

                const maxRetries = 5;

               

                while (retries < maxRetries) {

                    try {

                        const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, {

                            method: 'POST',

                            headers: { 'Content-Type': 'application/json' },

                            body: JSON.stringify({

                                contents: [{ parts: [{ text: prompt }] }],

                                systemInstruction: { parts: [{ text: systemInstruction }] }

                            })

                        });

                        const result = await response.json();

                        return result.candidates?.[0]?.content?.parts?.[0]?.text;

                    } catch (err) {

                        retries++;

                        await new Promise(r => setTimeout(r, Math.pow(2, retries) * 1000));

                    }

                }

                throw new Error("No s'ha pogut connectar amb la IA després de diversos intents.");

            };


            /**

             * Utilitza Gemini per millorar la redacció de la incidència

             */

            const refineDescription = async () => {

                if (!descripcio || descripcio.length < 5) return;

                setIsAiLoading(true);

                try {

                    const systemPrompt = "Ets un expert en suport informàtic escolar. Millora les notes de l'usuari i converteix-les en una descripció tècnica clara i professional en català. Sigues breu i directe.";

                    const refined = await callGemini(`Millora aquesta descripció d'incidència: ${descripcio}`, systemPrompt);

                    if (refined) setDescripcio(refined.trim());

                } catch (e) {

                    console.error("AI Error:", e);

                } finally {

                    setIsAiLoading(false);

                }

            };


            /**

             * Genera una possible solució ràpida per a una incidència existent

             */

            const getSolutionSuggestion = async (incidencia) => {

                setAiSuggestion({ id: incidencia.id, text: 'Analitzant el problema...' });

                try {

                    const systemPrompt = "Ets un coordinador TAC. Proposa 3 passos ràpids per intentar solucionar aquest problema abans que vingui el tècnic. Respon en català i de forma molt concisa.";

                    const prompt = `Incidència: ${incidencia.descripcio}. Tipus: ${incidencia.tipus}. Aula: ${incidencia.aula}.`;

                    const solution = await callGemini(prompt, systemPrompt);

                    setAiSuggestion({ id: incidencia.id, text: solution });

                } catch (e) {

                    setAiSuggestion({ id: incidencia.id, text: 'No s\'ha pogut generar el suggeriment.' });

                }

            };


            const handleImageCapture = (e) => {

                const file = e.target.files[0];

                if (file) {

                    const reader = new FileReader();

                    reader.onloadend = () => setImageData(reader.result);

                    reader.readAsDataURL(file);

                }

            };


            const exportToCSV = () => {

                if (incidencies.length === 0) return;

                const headers = ["Data", "Docent", "Ubicació", "Tipus", "Prioritat", "Estat", "Descripció"];

                const rows = incidencies.map(item => [

                    item.createdAt?.toDate().toLocaleDateString('ca-ES') || '',

                    `"${item.docent || 'Anònim'}"`,

                    `"${item.aula}"`,

                    item.tipus,

                    item.prioritat,

                    item.status,

                    `"${item.descripcio.replace(/"/g, '""')}"`

                ]);

                const csvContent = "\ufeff" + [headers, ...rows].map(e => e.join(",")).join("\n");

                const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });

                const url = URL.createObjectURL(blob);

                const link = document.createElement("a");

                link.href = url;

                link.download = `incidencies_tac_escola.csv`;

                link.click();

            };


            const handleSubmit = async (e) => {

                e.preventDefault();

                if (!descripcio || !aula || !docent || !user) return;

                setIsSubmitting(true);

                try {

                    await window.fb.addDoc(window.fb.collection(window.fb.db, 'artifacts', window.fb.appId, 'public', 'data', 'incidencies'), {

                        docent, descripcio, aula, prioritat, tipus, image: imageData,

                        status: 'pendent', userId: user.uid, createdAt: window.fb.Timestamp.now()

                    });

                    setDescripcio(''); setAula(''); setDocent(''); setImageData(null);

                    setView('list');

                } catch (err) { console.error(err); }

                finally { setIsSubmitting(false); }

            };


            const toggleStatus = async (id, currentStatus) => {

                const newStatus = currentStatus === 'pendent' ? 'resolta' : 'pendent';

                await window.fb.updateDoc(window.fb.doc(window.fb.db, 'artifacts', window.fb.appId, 'public', 'data', 'incidencies', id), { status: newStatus });

            };


            const deleteIncidencia = async (id) => {

                if (window.confirm('Estàs segur que vols eliminar aquesta incidència?')) {

                    await window.fb.deleteDoc(window.fb.doc(window.fb.db, 'artifacts', window.fb.appId, 'public', 'data', 'incidencies', id));

                }

            };


            if (loading) return (

                <div className="flex h-screen items-center justify-center bg-slate-50">

                    <div className="flex flex-col items-center gap-4 text-center">

                        <div className="w-12 h-12 border-4 border-blue-600 border-t-transparent rounded-full animate-spin"></div>

                        <p className="text-slate-500 font-semibold uppercase tracking-wider">Iniciant Gestor TAC...</p>

                    </div>

                </div>

            );


            return (

                <div className="min-h-screen pb-24 max-w-2xl mx-auto bg-white shadow-2xl relative">

                    {/* Capçalera */}

                    <header className="bg-blue-700 text-white p-5 sticky top-0 z-20 shadow-lg flex justify-between items-center">

                        <div className="flex items-center gap-3">

                            <div className="bg-white/20 p-2 rounded-lg">

                                <Icon name="Settings" size={24} className="text-white" />

                            </div>

                            <h1 className="text-xl font-black tracking-tight uppercase">Incidències TAC</h1>

                        </div>

                        <button onClick={exportToCSV} className="flex items-center gap-2 bg-blue-600 hover:bg-blue-500 px-4 py-2 rounded-xl text-xs font-bold transition-all active:scale-95 shadow-inner">

                            <Icon name="Download" size={16} /> EXPORTAR

                        </button>

                    </header>


                    <main className="p-4">

                        {view === 'list' ? (

                            <div className="space-y-4 animate-in">

                                <div className="flex justify-between items-end mb-2 px-1">

                                    <h2 className="text-lg font-bold text-slate-800 flex items-center gap-2">

                                        <Icon name="Layers" size={20} className="text-blue-600"/>

                                        Historial

                                    </h2>

                                    <span className="text-[10px] bg-blue-100 text-blue-700 px-3 py-1 rounded-full font-black uppercase">

                                        {incidencies.length} Registres

                                    </span>

                                </div>

                               

                                {incidencies.length === 0 ? (

                                    <div className="py-24 text-center flex flex-col items-center gap-6">

                                        <div className="bg-slate-100 p-8 rounded-full">

                                            <Icon name="SearchX" size={48} className="text-slate-300" />

                                        </div>

                                        <p className="text-slate-400 font-bold uppercase text-sm tracking-widest">Tot en ordre per ara</p>

                                    </div>

                                ) : incidencies.map(item => (

                                    <div key={item.id} className={`bg-white rounded-2xl shadow-sm border border-slate-200 border-l-[6px] p-5 transition-all ${item.status === 'resolta' ? 'border-green-500 opacity-60 grayscale-[0.5]' : 'border-amber-500'}`}>

                                        <div className="flex justify-between gap-4">

                                            <div className="flex-1 min-w-0">

                                                <div className="flex flex-wrap gap-2 mb-3">

                                                    <span className={`text-[10px] font-black px-3 py-1 rounded-lg uppercase ${item.status === 'resolta' ? 'bg-green-100 text-green-700' : 'bg-amber-100 text-amber-700'}`}>

                                                        {item.status}

                                                    </span>

                                                    <span className="text-[10px] font-black px-3 py-1 rounded-lg bg-slate-100 text-slate-600 uppercase">

                                                        {item.prioritat}

                                                    </span>

                                                </div>

                                                <h3 className="font-black text-xl text-slate-800 leading-tight mb-1">{item.aula}</h3>

                                                <div className="flex items-center gap-2 text-blue-600 text-xs font-bold mb-3 italic">

                                                    <Icon name="User" size={14} />

                                                    <span>{item.docent}</span>

                                                </div>

                                                <p className="text-slate-600 text-sm whitespace-pre-wrap leading-relaxed bg-slate-50 p-3 rounded-xl border border-slate-100">{item.descripcio}</p>

                                               

                                                {/* ✨ Secció IA Suggeriment */}

                                                <div className="mt-4">

                                                    {aiSuggestion.id === item.id ? (

                                                        <div className="bg-indigo-50 p-4 rounded-xl border border-indigo-100 text-xs text-indigo-900 animate-in relative overflow-hidden">

                                                            <div className="absolute top-0 right-0 p-2 opacity-10">

                                                                <Icon name="Sparkles" size={40} />

                                                            </div>

                                                            <div className="font-black flex items-center gap-2 mb-2 text-indigo-700 uppercase tracking-tighter">

                                                                <Icon name="Sparkles" size={14} /> Suggeriment IA

                                                            </div>

                                                            <p className="whitespace-pre-wrap font-medium leading-relaxed">{aiSuggestion.text}</p>

                                                        </div>

                                                    ) : (

                                                        <button

                                                            onClick={() => getSolutionSuggestion(item)}

                                                            className="text-[10px] font-black text-white bg-indigo-600 px-4 py-2 rounded-xl hover:bg-indigo-700 transition-all flex items-center gap-2 shadow-md active:scale-95"
